```c++
## 📂 대상 파일 / 모듈

## 🔍 리팩토링 전 (문제점)
- DestroySelf라는 이름이 풀 반환 동작을 제대로 표현 돼지 않았다
- OriginSize setter가 public이라 외부에서 자유롭게 변경 가능하다

## ✅ 리팩토링 후 (개선 내용)
- DestroySelf → ReturnPool로 메서드명을 변경 했다
- OriginSize에 setter를 private set 으로 변경 했다

## 💡 배운 점 / 메모
- 메서드 이름을 지을때 좀더 메서드의 기능과 역할을 잘 표현 할 수 있도록 네이밍 해야겠다.
```


```c#
## 개선전
public abstract class MinimapElement : MonoBehaviour, IPoolable
{
    [field:SerializeField] public bool SyncChildScale { get; set; }
    public string ID { get; set; }
    public RectTransform Rect { get; private set; }
    [field:SerializeField] public Vector2 NormalizedPos { get; set; }
    public Vector2 OriginSize { get; set; }

    protected virtual void Awake()
    {
        Rect = GetComponent<RectTransform>();
        OriginSize = Rect.sizeDelta;
    }

    public void RemoveSelf()
    {
        var evt = new RemoveMinimapElementEvent(ID);
        Bus.Raise(evt);
    }
    
    public void DestroySelf()
    {
        _myPool.Push(this);
    }

    #region Pool

    private Pool _myPool;
    [field:SerializeField] public PoolItemSO PoolItem { get; set; }
    public GameObject GameObject => gameObject;
    
    public void SetUpPool(Pool pool)
    {
        _myPool = pool;
    }

    public virtual void ResetItem()
    {
    }

    #endregion
}
```
